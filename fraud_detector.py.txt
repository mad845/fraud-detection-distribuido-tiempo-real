MOTOR 1: Reglas rápidas <10ms
def check_rules(transaction):
    last_loc = r.get(f"user:{transaction['user_id']}:last_loc")
    if last_loc and last_loc.decode() != transaction['location'] and transaction['amount'] > 50000:
        return 1 # Fraude por geolocalización imposible
    r.set(f"user:{transaction['user_id']}:last_loc", transaction['location'])
    return 0

MOTOR 2: ML Score
def ml_score(features):
    # features = [amount, hour, is_foreign_ip, etc]
    return model.predict_proba([features])[0][1]
